% -------------------- 参考文献交叉引用 --------------------
% 前文写 \CiteRef{文献键}；参考文献栏写

}

% -------------------- 可跨页正文单元格 --------------------
% 用户要求正文区左栏变窄：这里设为 48 bp，约能容纳两到三个五号汉字。
% 首页个人信息表仍保持原 Word 的列宽，不受这里影响。
\newcommand{\DrawRowDivider}{%
  \draw[line width=.48bp]
    ([xshift=48bp]frame.north west)--([xshift=48bp]frame.south west);%
}
\newcommand{\DrawFullRowLabel}[1]{%
  \node[align=center,text width=38.4bp,inner sep=0pt,outer sep=0pt]
    at ([xshift=24bp]frame.west) {{\LabelStyle #1}};%
}
% enhanced jigsaw 会在断页处省略横边框，因此在 overlay 中显式补画。
\newcommand{\DrawBreakTopRule}{%
  \draw[line width=.48bp] (frame.north west)--(frame.north east);%
}
\newcommand{\DrawBreakBottomRule}{%
  \draw[line width=.48bp] (frame.south west)--(frame.south east);%
}

% 先按加宽后的右栏真实宽度排一次内容，只测量高度而不输出。
% 测量时临时关闭 \label 写入，避免参考文献标签被重复写入辅助文件。
\newsavebox{\BnuRowMeasureBox}
\newsavebox{\BnuLabelMeasureBox}
\newdimen\BnuRowMeasuredHeight
\newdimen\BnuLabelMeasuredHeight
\newdimen\BnuMinimumRowHeight
\newcounter{bnumeasureref}
\newcommand{\MeasureReportRow}[1]{%
  \begingroup
    \renewcommand{\label}[1]{}%
    % 测量参考文献时保留真实的编号宽度和悬挂缩进，但不写入正式标签。
    \setcounter{bnumeasureref}{0}%
    \renewcommand{\ReferenceItem}[2]{%
      \stepcounter{bnumeasureref}%
      \par\noindent
      \hangindent=2.5em
      \hangafter=1
      [\arabic{bnumeasureref}]\ ##2\par
    }%
    % 图片在测量阶段照常占位，但不增加正式图号，也不写入标签。
    \renewcommand{\ReportFigure}[4][0.82\linewidth]{%
      \par\medskip
      \begin{center}
        \includegraphics[width=##1]{images/##3}\par
        \vspace{2pt}图~00\quad ##4\par
      \end{center}
      \medskip
    }%
    \setbox\BnuRowMeasureBox=\vbox{%
      \hsize=370.80bp
      \BodyStyle\justifying
      #1\par
    }%
    % 7.92 bp = 上下内边距与上下边框的总高度。
    \global\BnuRowMeasuredHeight=\dimexpr
      \ht\BnuRowMeasureBox+\dp\BnuRowMeasureBox+7.92bp\relax
  \endgroup
}

% 窄栏中的完整栏目名也参与高度测量，避免栏目名被裁切。
\newcommand{\MeasureRowLabel}[1]{%
  \begingroup
    \setbox\BnuLabelMeasureBox=\vbox{%
      \hsize=38.4bp
      \LabelStyle\centering
      #1\par
    }%
    \global\BnuLabelMeasuredHeight=\dimexpr
      \ht\BnuLabelMeasureBox+\dp\BnuLabelMeasureBox+7.92bp\relax
  \endgroup
}

% 所有正文行共用的边框与内边距。
% toprule/bottomrule at break 强制跨页处同时出现页底线和下一页页首线。
\tcbset{bnu report box/.style={%
  enhanced jigsaw,
  width=\linewidth,
  colback=white,
  colframe=black,
  boxrule=.48bp,
  toprule at break=.48bp,
  bottomrule at break=.48bp,
  topsep at break=3.48bp,
  bottomsep at break=3.48bp,
  sharp corners,
  boxsep=0pt,
  left=53.28bp,
  right=5.28bp,
  top=3.48bp,
  bottom=3.48bp,
  before skip=0pt,
  after skip=0pt,
  fontupper=\BodyStyle,
  fontlower=\BodyStyle
}}

% 参数：#1 完整栏目名；#2 原 Word 最小行高；#3 内容。
% 每一个跨页片段都重复完整栏目名，不再改成缩写或添加“（续）”。
\newcommand{\ReportRow}[3]{%
  \MeasureReportRow{#3}%
  \MeasureRowLabel{#1}%
  \BnuMinimumRowHeight=#2\relax
  \ifdim\BnuLabelMeasuredHeight>\BnuMinimumRowHeight
    \BnuMinimumRowHeight=\BnuLabelMeasuredHeight
  \fi
  \ifdim\BnuRowMeasuredHeight>\BnuMinimumRowHeight
    % 所有跨页片段都延伸到该页底部，保证完整栏目名有足够高度。
    \tcbset{bnu current row sizing/.style={%
      breakable,natural height,height fixed for=all}}%
  \else
    \tcbset{bnu current row sizing/.style={unbreakable,height=\BnuMinimumRowHeight}}%
  \fi
  \begin{tcolorbox}[
    bnu report box,
    bnu current row sizing,
    overlay unbroken={\DrawRowDivider\DrawFullRowLabel{#1}},
    overlay first={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakBottomRule},
    overlay middle={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakTopRule\DrawBreakBottomRule},
    overlay last={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakTopRule}
  ]
    \justifying #3\par
  \end{tcolorbox}%
}

% 意见栏使用 tcolorbox 的上下两部分：上部只放“意见：”和实际意见，
% 下部固定放审批、签字和日期。space to upper 把下部可靠地推到单元格底端。
\newcommand{\ReportOpinionRow}[4]{%
  \MeasureReportRow{#3\par\vspace{6pt}#4}%
  \MeasureRowLabel{#1}%
  \BnuMinimumRowHeight=#2\relax
  \ifdim\BnuLabelMeasuredHeight>\BnuMinimumRowHeight
    \BnuMinimumRowHeight=\BnuLabelMeasuredHeight
  \fi
  \ifdim\BnuRowMeasuredHeight>\BnuMinimumRowHeight
    \tcbset{bnu current row sizing/.style={%
      breakable,natural height,height fixed for=all}}%
  \else
    \tcbset{bnu current row sizing/.style={unbreakable,height=\BnuMinimumRowHeight}}%
  \fi
  \begin{tcolorbox}[
    bnu report box,
    bnu current row sizing,
    space to upper,
    lower separated=false,
    segmentation empty,
    middle=0pt,
    overlay unbroken={\DrawRowDivider\DrawFullRowLabel{#1}},
    overlay first={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakBottomRule},
    overlay middle={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakTopRule\DrawBreakBottomRule},
    overlay last={\DrawRowDivider\DrawFullRowLabel{#1}\DrawBreakTopRule}
  ]
    \justifying #3\par
    \tcblower
    #4\par
  \end{tcolorbox}%
}
